cmake_minimum_required(VERSION 3.16)
project(k210_greeter_core C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(ENABLE_SANITIZERS "Enable ASan and UBSan on host" OFF)
add_library(greeter_core core/greeter.c core/audio_queue.c)
target_include_directories(greeter_core PUBLIC core)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
 add_compile_options(-Wall -Wextra -Wpedantic -Werror)
 target_compile_options(greeter_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
 if(ENABLE_SANITIZERS)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
  target_compile_options(greeter_core PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
 endif()
endif()
if(NOT MSVC)
 target_link_libraries(greeter_core PUBLIC m)
endif()
enable_testing()
add_executable(core_tests tests/test_core.c)
target_link_libraries(core_tests PRIVATE greeter_core)
foreach(case normal reverse midpoint direct_jump foldback jitter dropout frame_gap loss ambiguous invalid chronology controls config timeout association axis overflow stable_gap grace_boundary evidence_jump stationary audio_fifo audio_retry audio_expiry audio_mute audio_clock audio_invalid audio_simulation)
 add_test(NAME ${case} COMMAND core_tests ${case})
endforeach()
add_library(greeter_replay SHARED tools/host/replay_bridge.c)
target_link_libraries(greeter_replay PRIVATE greeter_core)
set_target_properties(greeter_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME jsonl_replay COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_replay.py $<TARGET_FILE:greeter_replay>)
add_library(integration_example OBJECT tools/host/integration_example.c)
target_link_libraries(integration_example PRIVATE greeter_core)
# Python/ctypes replay is validated by the normal build. An ASan DSO cannot
# safely be loaded into an unsanitized Python interpreter without preloading.
if(ENABLE_SANITIZERS)
 set_tests_properties(jsonl_replay PROPERTIES DISABLED TRUE)
endif()

set_target_properties(greeter_replay PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS ON)
add_executable(resource_sizes tests/resource_sizes.c)
target_link_libraries(resource_sizes PRIVATE greeter_core)
add_test(NAME resource_sizes COMMAND resource_sizes)
